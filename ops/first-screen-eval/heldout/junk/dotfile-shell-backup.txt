mkcd() { mkdir -p "$1" && cd "$1"; }
extract() {
  case "$1" in
    *.tar.gz) tar xzf "$1" ;;
    *.zip)    unzip "$1" ;;
    *.7z)     7z x "$1" ;;
    *)        echo "don't know how to extract $1" ;;
  esac
}
