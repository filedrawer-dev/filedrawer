server {
    listen 80;
    server_name photos.cottage.example;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    server_name photos.cottage.example;

    ssl_certificate     /etc/ssl/cottage/fullchain.pem;
    ssl_certificate_key /etc/ssl/cottage/privkey.pem;

    client_max_body_size 2g;
    root /srv/photos/public;

    location / {
        proxy_pass http://127.0.0.1:2342;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_read_timeout 300s;
    }

    location /thumbs/ {
        alias /srv/photos/cache/thumbs/;
        expires 30d;
    }
}

# docker-compose.yml for the same box
services:
  photos:
    image: local/photo-gallery:stable
    restart: unless-stopped
    ports:
      - "127.0.0.1:2342:2342"
    volumes:
      - /srv/photos/originals:/originals
      - /srv/photos/cache:/cache
    environment:
      GALLERY_SITE_TITLE: "Cottage photos"
      GALLERY_READONLY: "false"
      GALLERY_ADMIN_PASSWORD: "CHANGE-ME-BEFORE-USE"
  backup:
    image: local/rsync-cron:latest
    volumes:
      - /srv/photos/originals:/data:ro
      - /mnt/usbdisk:/backup
    environment:
      CRON: "15 3 * * *"

# crontab -l
15 3 * * *  /usr/local/bin/backup-photos.sh >> /var/log/backup-photos.log 2>&1
0  4 * * 0  /usr/bin/docker system prune -f
