import csv
from collections import defaultdict

def load_invoices(path):
    totals = defaultdict(float)
    with open(path, newline="") as f:
        for row in csv.DictReader(f):
            totals[row["customer"]] += float(row["qty"]) * float(row["unit_price"])
    return totals

def print_report(totals):
    for customer, total in sorted(totals.items(), key=lambda kv: -kv[1]):
        print(f"{customer:<24}{total:>10.2f}")

if __name__ == "__main__":
    print_report(load_invoices("orders_sept.csv"))
